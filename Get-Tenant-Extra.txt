# ====================================================
# DSTA GOVERNANCE REVIEW - ADDITIONAL CSV EXPORTS
# ====================================================

Write-Host "Exporting Additional Governance Settings..."

# ----------------------------------------------------
# SHAREPOINT ONLINE - Sharing Controls
# ----------------------------------------------------
$SPOTenant |
    Select-Object `
        SharingCapability,
        ConditionalAccessPolicy,
        ShowEveryoneClaim,
        ShowEveryoneExceptExternalUsersClaim,
        DefaultLinkPermission,
        DefaultSharingLinkType |
    Export-Csv "$CsvPath\SPO-SharingSettings.csv" -NoTypeInformation

# ----------------------------------------------------
# CONDITIONAL ACCESS
# ----------------------------------------------------
try
{
    $CAPolicies = Get-MgIdentityConditionalAccessPolicy

    $CAPolicies |
        Select-Object `
            DisplayName,
            State,
            CreatedDateTime,
            ModifiedDateTime |
        Export-Csv "$CsvPath\ConditionalAccessPolicies.csv" -NoTypeInformation
}
catch
{
    Write-Warning "Conditional Access export skipped"
}

# ----------------------------------------------------
# AUTHENTICATION METHODS
# ----------------------------------------------------
try
{
    $AuthPolicy = Get-MgPolicyAuthenticationMethodPolicy

    $AuthPolicy |
        Select-Object Id, PolicyMigrationState |
        Export-Csv "$CsvPath\AuthMethodPolicy.csv" -NoTypeInformation
}
catch
{
    Write-Warning "Authentication Method export skipped"
}

# ----------------------------------------------------
# POWER PLATFORM TENANT SETTINGS
# ----------------------------------------------------
$TenantSettings.PSObject.Properties |
    Select-Object Name, Value |
    Export-Csv "$CsvPath\PowerPlatform-TenantSettings.csv" -NoTypeInformation
